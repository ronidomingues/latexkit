% A versao curta (entre colchetes) e a que aparece no rodape; sem ela, o
% beamer usa o titulo completo.
\lkIfShortTitle{\title[\lkShortTitle]{\lkTitle}}{\title{\lkTitle}}
\lkIfSubtitle{\subtitle{\lkSubtitle}}{}
\author{\lkAuthor}
\institute{%
  \lkIfInstitution{\lkInstitution}{}%
  \lkIfEvent{\\[0.3em]\small\lkEvent}{}%
}
% Data vazia significa "data da compilacao".
\lkIfDate{\date{\lkDate}}{\date{\today}}

\hypersetup{
  pdftitle={\lkTitle},
  pdfauthor={\lkAuthor},
  pdfcreator={latexkit}
}
