  \lgIfInstitution{\lgInstitution}{}%
  \lgIfEvent{\\[0.3em]\small\lgEvent}{}%
}

  pdftitle={\lgTitle},
  pdfauthor={\lgAuthor},
  pdfcreator={latexgen}
}
