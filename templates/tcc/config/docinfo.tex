% =====================================================================
%  Liga os metadados do latexgen.config.json aos comandos da abntex2.
%
%  As macros \lgAlgumaCoisa vem de config/metadata.tex, que e gerado.
%  \lgIfAlgumaCoisa{ha valor}{esta vazio} permite omitir blocos inteiros
%  quando o campo nao foi preenchido — e o que faz o coorientador sumir
%  da folha de rosto quando nao existe.
%
%  Normalmente nao ha motivo para editar este arquivo: para mudar um
%  valor, edite latexgen.config.json; para mudar a aparencia, use
%  config/style.tex.

\titulo{\lgTitle}
\autor{\lgAuthor}
\local{\lgCity}
\data{\lgYear}
\orientador{\lgAdvisor}
\lgIfCoadvisor{\coorientador{\lgCoadvisor}}{}
\tipotrabalho{\lgDegreeType}

% Instituicao: a unidade (faculdade, instituto, centro) entra em uma
% segunda linha, quando informada.
\instituicao{%
  \lgInstitution
  \lgIfUnit{\par\lgUnit}{}%
  \lgIfCourse{\par\lgCourse}{}%
}

% Preambulo: o paragrafo da folha de rosto que declara a que o trabalho
% se destina. A NBR 14724 exige a natureza do trabalho, o objetivo, o
% nome da instituicao e a area de concentracao. Cada instituicao tem a
% sua redacao exata — a sua esta em latexgen.config.json, no campo
% "preamble".
\preambulo{\lgPreamble}

% Metadados do PDF, usados por leitores e buscadores.
\makeatletter
\hypersetup{
  pdftitle={\@title},
  pdfauthor={\@author},
  pdfsubject={\lgPreamble},
  pdfkeywords={\lgKeywords},
  pdfcreator={latexgen}
}
\makeatother
